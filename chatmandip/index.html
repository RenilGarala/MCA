(user.name + ' (Logout)') : 'Login'; authLink.href = user ? '#' : 'login.html'; if(user){ authLink.addEventListener('click', (e)=>{ e.preventDefault(); logout(); }); } } // Signup const signupForm = document.getElementById('signupForm'); if(signupForm){ signupForm.addEventListener('submit', e=>{ e.preventDefault(); const name=document.getElementById('sname').value.trim(); const email=document.getElementById('semail').value.trim(); const pass=document.getElementById('spassword').value; if(!name||!email||!pass) return; // simple users store const users = JSON.parse(localStorage.getItem('eg_users')||'[]'); if(users.find(u=>u.email===email)){ document.getElementById('signupMsg').textContent='Email already registered.';return; } users.push({name, email, pass}); localStorage.setItem('eg_users', JSON.stringify(users)); setUser({name,email}); document.getElementById('signupMsg').textContent='Account created! Redirecting...'; setTimeout(()=>location.href='index.html',800); }); } // Login const loginForm = document.getElementById('loginForm'); if(loginForm){ loginForm.addEventListener('submit', e=>{ e.preventDefault(); const email=document.getElementById('email').value.trim(); const pass=document.getElementById('password').value; const users = JSON.parse(localStorage.getItem('eg_users')||'[]'); const u = users.find(x=>x.email===email && x.pass===pass); if(!u){ document.getElementById('loginMsg').textContent='Invalid credentials'; return; } setUser({name:u.name,email:u.email}); document.getElementById('loginMsg').textContent='Logged in! Redirecting...'; setTimeout(()=>location.href='index.html',700); }); } ``` --- ## `js/trainers.js` ```javascript const membershipForm = document.getElementById('membershipForm'); if(membershipForm){ membershipForm.addEventListener('submit', e=>{ e.preventDefault(); const user = JSON.parse(localStorage.getItem('eg_user')||'null'); if(!user){ alert('Please login or signup first.'); location.href='login.html'; return; } const type = document.getElementById('membershipType').value; const days = Array.from(document.querySelectorAll('input[name="days"]:checked')).map(i=>i.value); if(days.length===0){ document.getElementById('membershipMsg').textContent='Select at least one training day.'; return; } // store membership per user const members = JSON.parse(localStorage.getItem('eg_members')||'{}'); members[user.email] = {type, days, since: new Date().toISOString()}; localStorage.setItem('eg_members', JSON.stringify(members)); document.getElementById('membershipMsg').textContent = 'Membership saved.
